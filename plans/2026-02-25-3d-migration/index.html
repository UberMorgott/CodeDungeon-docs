    2D → 3D Migration Plan¶

    For Claude: REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

    Goal: Replace the 2D isometric rendering with real 3D (orthogonal camera, GLB models, AnimationTree) while keeping the server and UI layers untouched.

    Architecture: Server still outputs flat (x, y) coordinates. Client converts them to Vector3(x * SCALE, 0, y * SCALE) where SCALE = 0.01, giving a 12m x 10m room. Camera3D in orthogonal mode at iso angle (-35° X, -45° Y) reproduces the same Diablo/PoE feel. All UI (Control-based screens) remains 2D.

    Tech Stack: Godot 4.6, GDScript, GLB models (KayKit Adventurers CC0), AnimationTree + AnimationNodeStateMachine.

    Task 3: Create character_3d.gd Script¶

    Files: - Create: roguelite/scripts/entities/character_3d.gd

    This is the main script. It replaces character.gd entirely. Key differences: - Positions are Vector3 (converted from server x, y) - Model rotation replaces sprite direction rows - GLB models loaded per entity type instead of sprite sheets

    Step 1: Write the script

    extends Node3D
## 3D visual representation of a simulation entity.
## No physics — position set directly from server snapshots.

@onready var model_pivot: Node3D = $Model
@onready var name_label: Label3D = $NameLabel
@onready var hp_bar: Label3D = $HPBar
@onready var ai_debug_label: Label3D = $AIDebugLabel

var entity_id: String = ""
var entity_type: int = 0

## Server (x, y) → Godot Vector3(x, 0, y) scale factor.
## Server room is 1200x1000 units → 12m x 10m in 3D.
const SCALE := 0.01

var _prev_server_x: float = 0.0
var _prev_server_y: float = 0.0
var _target_rotation_y: float = 0.0
var _current_action: int = 0

# Model scenes — preloaded per entity type.
# Adjust paths after importing GLB files.
var _model_scene_player: PackedScene = preload("res://assets/models/kaykit/Barbarian.glb")
var _model_scene_melee: PackedScene = preload("res://assets/models/kaykit/Skeleton.glb")
var _model_scene_ranged: PackedScene = preload("res://assets/models/kaykit/SkeletonMage.glb")

# Protojson enum string -> int mappings (same as character.gd).
const ACTION_MAP := {
    "ACTION_IDLE": 0, "ACTION_MOVE": 1, "ACTION_ATTACK": 2,
    "ACTION_HIT": 3, "ACTION_DEAD": 4, "ACTION_CAST": 5,
}

const NODE_DISPLAY_NAMES := {
    "NODE_UNSPECIFIED": "?",
    "NODE_PRIORITY_SELECTOR": "Priority",
    "NODE_SEQUENCE": "Sequence",
    "NODE_HP_CHECK": "HPCheck",
    "NODE_DISTANCE_CHECK": "DistCheck",
    "NODE_MOVE_TO": "MoveTo",
    "NODE_FLEE_FROM": "Flee",
    "NODE_USE_SKILL": "UseSkill",
    "NODE_WAIT": "Wait",
}


static func parse_action(raw) -> int:
    if raw is int or raw is float:
        return int(raw)
    return ACTION_MAP.get(str(raw), 0)


static func server_to_world(sx: float, sy: float) -> Vector3:
    ## Convert server (x, y) to Godot 3D world position.
    return Vector3(sx * SCALE, 0.0, sy * SCALE)


func setup(id: String, type_val: int, display_name: String = "") -> void:
    entity_id = id
    entity_type = type_val
    name_label.text = display_name if display_name != "" else id

    # Clear any previous model.
    for child in model_pivot.get_children():
        child.queue_free()

    match type_val:
        1:  # PLAYER
            _instantiate_model(_model_scene_player)
        2:  # ENEMY_MELEE
            _instantiate_model(_model_scene_melee)
        3:  # ENEMY_RANGED
            _instantiate_model(_model_scene_ranged)
        4:  # PROJECTILE
            _setup_projectile()
            return
        _:
            _instantiate_model(_model_scene_player)


func _instantiate_model(scene: PackedScene) -> void:
    var instance := scene.instantiate()
    model_pivot.add_child(instance)
    # KayKit models may need scale adjustment — tweak after visual test.
    # instance.scale = Vector3(0.5, 0.5, 0.5)


func _setup_projectile() -> void:
    ## Simple sphere for projectiles.
    var mesh_inst := MeshInstance3D.new()
    var sphere := SphereMesh.new()
    sphere.radius = 0.15
    sphere.height = 0.3
    mesh_inst.mesh = sphere
    var mat := StandardMaterial3D.new()
    mat.albedo_color = Color.YELLOW
    mat.emission_enabled = true
    mat.emission = Color.YELLOW
    mat.emission_energy_multiplier = 2.0
    mesh_inst.material_override = mat
    model_pivot.add_child(mesh_inst)
    # Elevate projectile to waist height.
    mesh_inst.position.y = 0.8
    name_label.visible = false
    hp_bar.visible = false


func apply_state(state: Dictionary) -> void:
    var sx: float = state.get("x", 0.0)
    var sy: float = state.get("y", 0.0)
    var world_pos := server_to_world(sx, sy)
    position = world_pos

    # Projectiles: just position, no rotation/HP.
    if entity_type == 4:
        return

    # Rotation from movement delta.
    var dx: float = sx - _prev_server_x
    var dy: float = sy - _prev_server_y
    if absf(dx) > 0.5 or absf(dy) > 0.5:
        _target_rotation_y = atan2(dx, dy)
    _prev_server_x = sx
    _prev_server_y = sy

    # Smooth rotation.
    model_pivot.rotation.y = lerp_angle(model_pivot.rotation.y, _target_rotation_y, 0.3)

    # HP bar (text-based for now).
    var hp: float = state.get("health", 0)
    var max_hp: float = state.get("maxHealth", 1)
    var ratio: float = hp / max_hp if max_hp > 0 else 0.0
    var filled: int = int(ratio * 10)
    hp_bar.text = "█".repeat(filled) + "░".repeat(10 - filled)
    hp_bar.modulate = Color(
        lerpf(0.8, 0.2, ratio),  # red when low
        lerpf(0.2, 0.8, ratio),  # green when high
        0.2, 0.9
    )
    hp_bar.visible = hp > 0

    # Action feedback.
    var action: int = parse_action(state.get("action", 0))
    _update_visuals(action)
    _current_action = action

    # AI debug overlay.
    _update_ai_debug(state)


func _update_visuals(action: int) -> void:
    ## Visual feedback per action. Phase A: simple modulate.
    ## Phase B replaces this with AnimationTree state machine.
    match action:
        4:  # ACTION_DEAD
            _set_model_modulate(Color(0.5, 0.5, 0.5, 0.5))
            hp_bar.visible = false
        2, 5:  # ACTION_ATTACK, ACTION_CAST
            _set_model_modulate(Color(1.2, 0.8, 0.8, 1.0))
        3:  # ACTION_HIT
            _set_model_modulate(Color(1.0, 0.5, 0.5, 1.0))
        _:  # IDLE, MOVE
            _set_model_modulate(Color(1, 1, 1, 1))


func _set_model_modulate(color: Color) -> void:
    ## Apply color tint to all MeshInstance3D children.
    ## Simple approach — finds meshes recursively in model.
    for mesh in _find_meshes(model_pivot):
        mesh.modulate = color  # Label3D/Sprite3D
        # For MeshInstance3D, use material override:
        if mesh is MeshInstance3D:
            var mat := mesh.get_active_material(0)
            if mat is StandardMaterial3D:
                mat.albedo_color = color


func _find_meshes(node: Node) -> Array:
    var result: Array = []
    if node is MeshInstance3D:
        result.append(node)
    for child in node.get_children():
        result.append_array(_find_meshes(child))
    return result


func _update_ai_debug(state: Dictionary) -> void:
    if entity_type != 1 or not SettingsManager.show_ai_decisions:
        ai_debug_label.visible = false
        return

    var module_name: String = state.get("activeModule", "")
    var node_type: String = state.get("activeNode", "")
    var skill: String = state.get("activeSkill", "")
    var node_short: String = NODE_DISPLAY_NAMES.get(node_type, node_type)

    var parts: PackedStringArray = []
    if module_name != "":
        parts.append(module_name)
    if skill != "":
        parts.append(node_short + ": " + skill)
    elif node_short != "":
        parts.append(node_short)

    if parts.is_empty():
        ai_debug_label.visible = false
        return

    ai_debug_label.text = " | ".join(parts)
    ai_debug_label.visible = true

    Step 2: Write game_world_3d.gd

    extends Node3D
## 3D game world — camera follow, zoom, environment.

@onready var camera: Camera3D = $Camera3D
@onready var entities_node: Node3D = $Entities

## Camera offset from target in world space.
## Computed from the iso angle so camera orbits correctly.
var _camera_offset: Vector3
var _camera_target: Vector3 = Vector3(6, 0, 5)  # room center default
var _camera_free: bool = false
var _camera_drag: bool = false
var _last_mouse_pos: Vector2 = Vector2.ZERO

const CAMERA_SMOOTH_SPEED := 5.0
const ZOOM_STEP := 0.5
const ZOOM_MIN := 8.0
const ZOOM_MAX := 30.0


func _ready() -> void:
    # Store initial offset so we can reposition camera relative to target.
    _camera_offset = camera.position - _camera_target


func _process(delta: float) -> void:
    if _camera_free:
        return
    # Smooth follow.
    var desired := _camera_target + _camera_offset
    camera.position = camera.position.lerp(desired, delta * CAMERA_SMOOTH_SPEED)


func follow_target(world_pos: Vector3) -> void:
    ## Called each tick to update camera target (e.g. player position).
    if not _camera_free:
        _camera_target = world_pos


func set_free_mode(free: bool) -> void:
    _camera_free = free


func zoom_in() -> void:
    camera.size = clampf(camera.size - ZOOM_STEP, ZOOM_MIN, ZOOM_MAX)


func zoom_out() -> void:
    camera.size = clampf(camera.size + ZOOM_STEP, ZOOM_MIN, ZOOM_MAX)


func snap_to_target() -> void:
    _camera_free = false
    camera.position = _camera_target + _camera_offset

    Step 2: Add AnimationTree to character_3d.tscn

    Add as sibling of Model node:

    AnimationTree properties:

    tree_root = AnimationNodeStateMachine (created in Step 3)

    Step 3: Build the state machine

    Create the AnimationNodeStateMachine programmatically in _ready() or via scene setup:

    StateMachine:
├── Idle         → AnimationNodeAnimation("Idle")
├── Locomotion   → AnimationNodeBlendSpace2D (see Task 10)
├── Attack       → AnimationNodeAnimation("Attack1")
├── Death        → AnimationNodeAnimation("Death")
└── Hit          → AnimationNodeAnimation("HitRecieve")

Transitions:
- Idle → Locomotion:  advance_condition = "is_moving"
- Locomotion → Idle:  advance_condition = "is_idle"
- * → Attack:         advance_condition = "is_attacking", switch_mode = AT_END
- * → Death:          advance_condition = "is_dead", switch_mode = IMMEDIATE
- * → Hit:            advance_condition = "is_hit", switch_mode = IMMEDIATE
- Attack → Idle:      auto_advance = true (after animation ends)
- Hit → Idle:         auto_advance = true (after animation ends)

    In GDScript (inside _ready() or a setup method, after model is instantiated):

    func _setup_animation_tree() -> void:
    # Find AnimationPlayer in the instantiated GLB model.
    var anim_player: AnimationPlayer = _find_child_of_type(model_pivot, "AnimationPlayer")
    if not anim_player:
        push_warning("No AnimationPlayer found in model for %s" % entity_id)
        return

    var anim_tree := AnimationTree.new()
    add_child(anim_tree)
    anim_tree.anim_player = anim_tree.get_path_to(anim_player)

    var sm := AnimationNodeStateMachine.new()

    # Add states.
    var idle_node := AnimationNodeAnimation.new()
    idle_node.animation = &"Idle"  # Verify exact name from GLB
    sm.add_node(&"Idle", idle_node, Vector2(0, 0))

    var attack_node := AnimationNodeAnimation.new()
    attack_node.animation = &"Attack1"
    sm.add_node(&"Attack", attack_node, Vector2(200, 0))

    var death_node := AnimationNodeAnimation.new()
    death_node.animation = &"Death"
    sm.add_node(&"Death", death_node, Vector2(200, 200))

    var hit_node := AnimationNodeAnimation.new()
    hit_node.animation = &"HitRecieve"
    sm.add_node(&"Hit", hit_node, Vector2(0, 200))

    # Locomotion — simple animation for now, BlendSpace2D in Task 10.
    var walk_node := AnimationNodeAnimation.new()
    walk_node.animation = &"Walk"
    sm.add_node(&"Locomotion", walk_node, Vector2(-200, 0))

    # Transitions.
    _add_transition(sm, &"Idle", &"Locomotion", "is_moving")
    _add_transition(sm, &"Locomotion", &"Idle", "is_idle")
    _add_transition(sm, &"Idle", &"Attack", "is_attacking")
    _add_transition(sm, &"Locomotion", &"Attack", "is_attacking")
    _add_transition(sm, &"Idle", &"Death", "is_dead")
    _add_transition(sm, &"Locomotion", &"Death", "is_dead")

    # Attack → Idle: auto-advance after animation finishes.
    var t_attack_idle := AnimationNodeStateMachineTransition.new()
    t_attack_idle.advance_mode = AnimationNodeStateMachineTransition.ADVANCE_MODE_AUTO
    sm.add_transition(&"Attack", &"Idle", t_attack_idle)

    # Hit → Idle: auto-advance.
    var t_hit_idle := AnimationNodeStateMachineTransition.new()
    t_hit_idle.advance_mode = AnimationNodeStateMachineTransition.ADVANCE_MODE_AUTO
    sm.add_transition(&"Hit", &"Idle", t_hit_idle)

    anim_tree.tree_root = sm
    anim_tree.active = true
    _anim_tree = anim_tree


func _add_transition(sm: AnimationNodeStateMachine, from: StringName, to: StringName, condition: String) -> void:
    var t := AnimationNodeStateMachineTransition.new()
    t.advance_condition = condition
    sm.add_transition(from, to, t)

    Reference: Server Coordinate Mapping¶

    The server operates in 2D (x, y) space with room size 1200 x 1000. The conversion:

    Server (x=0..1200, y=0..1000)
    ↓  × 0.01
Godot 3D (x=0..12, y=0, z=0..10)

    Camera is orthogonal at iso angle — rotation_degrees = (-35, -45, 0). This means: - Moving right on server (+x) moves right-and-toward-camera in 3D - Moving down on server (+y) moves left-and-toward-camera in 3D - Combined effect = classic isometric diamond view

    Model rotation uses atan2(dx, dz) where dx and dz are server deltas scaled to 3D. The model faces the direction of movement.
