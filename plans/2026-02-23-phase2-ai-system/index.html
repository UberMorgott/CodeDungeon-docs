    Phase 2: AI System — Implementation Plan¶

    For Claude: REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

    Goal: Replace hardcoded AI with a behavior tree system where player builds define entity behavior through blueprints, modules, and memory allocation.

    Architecture: Protobuf-defined blueprint schema → server-side behavior tree executor (internal/ai/ package) → module system with priority-based activation → memory cost validation. Client sends serialized PlayerBuild via UploadBuild; server parses it into a Node tree and runs it each tick instead of hardcoded if/else. Default blueprint replicates current hardcoded behavior for backward compat.

    Tech Stack: Go 1.25, Protobuf + protojson, Godot 4.6 GDScript, gorilla/websocket

    package ai

import pb "roguelite-server/pkg/proto"

// Status is the result of evaluating a behavior tree node.
type Status int

const (
    Success Status = iota
    Failure
)

// Action is what an entity should do this tick.
type Action struct {
    Type     pb.EntityAction
    TargetID string
}

// Context carries per-tick evaluation state through the tree.
type Context struct {
    Self     *Entity
    Entities []*Entity
    Target   *Entity // set by targeting nodes, read by action nodes
    Tick     int
}

// Entity is the AI-visible view of a game entity.
type Entity struct {
    ID     string
    Type   pb.EntityType
    X, Y   float64
    HP     int32
    MaxHP  int32
    Armor  float64
    Damage int32
    Speed  float64
    Alive  bool

    Cooldown int // ticks until next attack
}

// Node is a single behavior tree node.
type Node interface {
    Evaluate(ctx *Context) (Status, *Action)
}

    package ai

// PrioritySelector evaluates children top-to-bottom, returns first success.
type PrioritySelector struct {
    Children []Node
}

func (n *PrioritySelector) Evaluate(ctx *Context) (Status, *Action) {
    for _, child := range n.Children {
        s, a := child.Evaluate(ctx)
        if s == Success {
            return Success, a
        }
    }
    return Failure, nil
}

// Sequence evaluates children in order, fails on first failure.
// Returns the last successful child's action.
type Sequence struct {
    Children []Node
}

func (n *Sequence) Evaluate(ctx *Context) (Status, *Action) {
    var lastAction *Action
    for _, child := range n.Children {
        s, a := child.Evaluate(ctx)
        if s == Failure {
            return Failure, nil
        }
        if a != nil {
            lastAction = a
        }
    }
    return Success, lastAction
}

    package ai

import (
    "math"

    pb "roguelite-server/pkg/proto"
)

// HPBelow succeeds when target's HP is below threshold percent.
type HPBelow struct {
    Threshold float64 // 0-100
    Target    string  // "self" (future: "ally", "enemy")
}

func (n *HPBelow) Evaluate(ctx *Context) (Status, *Action) {
    var e *Entity
    switch n.Target {
    case "self", "":
        e = ctx.Self
    default:
        e = ctx.Self
    }
    pct := float64(e.HP) / float64(e.MaxHP) * 100
    if pct < n.Threshold {
        return Success, nil
    }
    return Failure, nil
}

// EnemyInRange succeeds if any living enemy is within Range pixels.
type EnemyInRange struct {
    Range float64
}

func (n *EnemyInRange) Evaluate(ctx *Context) (Status, *Action) {
    for _, e := range ctx.Entities {
        if e == ctx.Self || !e.Alive || !isEnemy(ctx.Self, e) {
            continue
        }
        if dist(ctx.Self, e) <= n.Range {
            return Success, nil
        }
    }
    return Failure, nil
}

func isEnemy(self, other *Entity) bool {
    selfIsPlayer := self.Type == pb.EntityType_PLAYER
    otherIsPlayer := other.Type == pb.EntityType_PLAYER
    return selfIsPlayer != otherIsPlayer
}

func dist(a, b *Entity) float64 {
    dx := a.X - b.X
    dy := a.Y - b.Y
    return math.Sqrt(dx*dx + dy*dy)
}

    Step 5: Implement targeting nodes¶

    package ai

import "math"

// NearestEnemy sets ctx.Target to the closest living enemy.
type NearestEnemy struct{}

func (n *NearestEnemy) Evaluate(ctx *Context) (Status, *Action) {
    var best *Entity
    bestDist := math.MaxFloat64
    for _, e := range ctx.Entities {
        if e == ctx.Self || !e.Alive || !isEnemy(ctx.Self, e) {
            continue
        }
        d := dist(ctx.Self, e)
        if d < bestDist {
            bestDist = d
            best = e
        }
    }
    if best == nil {
        return Failure, nil
    }
    ctx.Target = best
    return Success, nil
}

// LowestHPEnemy sets ctx.Target to the living enemy with lowest HP.
type LowestHPEnemy struct{}

func (n *LowestHPEnemy) Evaluate(ctx *Context) (Status, *Action) {
    var best *Entity
    for _, e := range ctx.Entities {
        if e == ctx.Self || !e.Alive || !isEnemy(ctx.Self, e) {
            continue
        }
        if best == nil || e.HP < best.HP {
            best = e
        }
    }
    if best == nil {
        return Failure, nil
    }
    ctx.Target = best
    return Success, nil
}

import pb "roguelite-server/pkg/proto"

// MoveTo returns a MOVE action toward ctx.Target.
type MoveTo struct{}

func (n *MoveTo) Evaluate(ctx *Context) (Status, *Action) {
    if ctx.Target == nil {
        return Failure, nil
    }
    return Success, &Action{Type: pb.EntityAction_MOVE, TargetID: ctx.Target.ID}
}

// UseSkill returns an ATTACK action on ctx.Target if not on cooldown.
// Phase 2: skill = basic attack. Phase 3: skill lookup by name/tag.
type UseSkill struct{}

func (n *UseSkill) Evaluate(ctx *Context) (Status, *Action) {
    if ctx.Target == nil || ctx.Self.Cooldown > 0 {
        return Failure, nil
    }
    return Success, &Action{Type: pb.EntityAction_ATTACK, TargetID: ctx.Target.ID}
}

// Wait returns an IDLE action.
type Wait struct{}

func (n *Wait) Evaluate(ctx *Context) (Status, *Action) {
    return Success, &Action{Type: pb.EntityAction_IDLE}
}

    package ai

import (
    "fmt"
    "strconv"

    pb "roguelite-server/pkg/proto"
)

// BuildTree converts a protobuf BlueprintNode into an executable Node tree.
// Returns the default behavior tree if proto is nil.
func BuildTree(proto *pb.BlueprintNode) (Node, error) {
    if proto == nil {
        return DefaultTree(), nil
    }
    return buildNode(proto)
}

// DefaultTree returns the fallback AI: find nearest enemy, attack or move.
func DefaultTree() Node {
    return &PrioritySelector{Children: []Node{
        &Sequence{Children: []Node{
            &NearestEnemy{},
            &EnemyInRange{Range: 40},
            &UseSkill{},
        }},
        &Sequence{Children: []Node{
            &NearestEnemy{},
            &MoveTo{},
        }},
    }}
}

func buildNode(proto *pb.BlueprintNode) (Node, error) {
    switch proto.Type {
    case pb.NodeType_NODE_PRIORITY_SELECTOR:
        children, err := buildChildren(proto.Children)
        if err != nil {
            return nil, err
        }
        return &PrioritySelector{Children: children}, nil

    case pb.NodeType_NODE_SEQUENCE:
        children, err := buildChildren(proto.Children)
        if err != nil {
            return nil, err
        }
        return &Sequence{Children: children}, nil

    case pb.NodeType_NODE_HP_BELOW:
        threshold := paramFloat(proto.Parameters, "threshold", 50)
        target := paramStr(proto.Parameters, "target", "self")
        return &HPBelow{Threshold: threshold, Target: target}, nil

    case pb.NodeType_NODE_ENEMY_IN_RANGE:
        r := paramFloat(proto.Parameters, "range", 40)
        return &EnemyInRange{Range: r}, nil

    case pb.NodeType_NODE_MOVE_TO:
        return &MoveTo{}, nil

    case pb.NodeType_NODE_USE_SKILL:
        return &UseSkill{}, nil

    case pb.NodeType_NODE_NEAREST_ENEMY:
        return &NearestEnemy{}, nil

    case pb.NodeType_NODE_LOWEST_HP_ENEMY:
        return &LowestHPEnemy{}, nil

    case pb.NodeType_NODE_WAIT:
        return &Wait{}, nil

    default:
        return nil, fmt.Errorf("unknown node type: %d", proto.Type)
    }
}

func buildChildren(protos []*pb.BlueprintNode) ([]Node, error) {
    nodes := make([]Node, 0, len(protos))
    for _, p := range protos {
        n, err := buildNode(p)
        if err != nil {
            return nil, err
        }
        nodes = append(nodes, n)
    }
    return nodes, nil
}

func paramFloat(params map[string]string, key string, fallback float64) float64 {
    v, ok := params[key]
    if !ok {
        return fallback
    }
    f, err := strconv.ParseFloat(v, 64)
    if err != nil {
        return fallback
    }
    return f
}

func paramStr(params map[string]string, key string, fallback string) string {
    v, ok := params[key]
    if !ok {
        return fallback
    }
    return v
}

    package ai

import "sort"

// ExecutorModule is a runtime module with pre-built condition and behavior trees.
type ExecutorModule struct {
    Name      string
    Priority  int // 1 = highest
    Condition Node
    Tree      Node
}

// Executor runs the AI for one entity per tick.
// Evaluates modules by priority, falls back to global AI.
type Executor struct {
    Global  Node
    Modules []*ExecutorModule // sorted by priority ascending (1 first)
}

// NewExecutor creates an executor with sorted modules.
func NewExecutor(global Node, modules []*ExecutorModule) *Executor {
    sorted := make([]*ExecutorModule, len(modules))
    copy(sorted, modules)
    sort.Slice(sorted, func(i, j int) bool {
        return sorted[i].Priority < sorted[j].Priority
    })
    return &Executor{Global: global, Modules: sorted}
}

// Evaluate runs the AI decision for one tick.
// Checks modules by priority, uses first whose condition succeeds.
// Falls back to global AI if no module activates.
func (e *Executor) Evaluate(ctx *Context) (Status, *Action) {
    for _, mod := range e.Modules {
        s, _ := mod.Condition.Evaluate(ctx)
        if s == Success {
            return mod.Tree.Evaluate(ctx)
        }
    }
    return e.Global.Evaluate(ctx)
}

    package ai

import (
    "fmt"

    pb "roguelite-server/pkg/proto"
)

// Node memory costs from GDD.
var nodeCosts = map[pb.NodeType]int32{
    pb.NodeType_NODE_HP_BELOW:           1,
    pb.NodeType_NODE_MOVE_TO:            1,
    pb.NodeType_NODE_SEQUENCE:           1,
    pb.NodeType_NODE_NEAREST_ENEMY:      1,
    pb.NodeType_NODE_LOWEST_HP_ENEMY:    1,
    pb.NodeType_NODE_WAIT:               1,
    pb.NodeType_NODE_USE_SKILL:          2,
    pb.NodeType_NODE_PRIORITY_SELECTOR:  2,
    pb.NodeType_NODE_ENEMY_IN_RANGE:     2,
}

// NodeCost returns the memory cost for a single node type.
func NodeCost(t pb.NodeType) int32 {
    if c, ok := nodeCosts[t]; ok {
        return c
    }
    return 1
}

// TreeCost returns total memory cost of a blueprint tree.
func TreeCost(node *pb.BlueprintNode) int32 {
    if node == nil {
        return 0
    }
    cost := NodeCost(node.Type)
    for _, child := range node.Children {
        cost += TreeCost(child)
    }
    return cost
}

// ValidateBuild checks that a player build fits within memory limits.
func ValidateBuild(build *pb.PlayerBuild) error {
    if build == nil {
        return nil // nil build = use defaults
    }

    globalCost := TreeCost(build.GlobalAi)
    if globalCost > build.GlobalMemory {
        return fmt.Errorf("global AI cost %d exceeds allocated memory %d", globalCost, build.GlobalMemory)
    }

    moduleMemory := build.TotalMemory - build.GlobalMemory
    var moduleCost int32
    for _, mod := range build.Modules {
        moduleCost += TreeCost(mod.Condition)
        moduleCost += TreeCost(mod.Blueprint)
    }
    if moduleCost > moduleMemory {
        return fmt.Errorf("modules cost %d exceeds allocated memory %d", moduleCost, moduleMemory)
    }

    return nil
}
