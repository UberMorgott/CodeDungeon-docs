    Phase 3a: Combat Foundation — Implementation Plan¶

    For Claude: REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

    Goal: Replace the hardcoded single-attack damage system with a full combat foundation: typed skills with individual cooldowns, GDD-accurate damage formula (crit, dodge, block, resistance), projectile entities, and enemy variety. After this phase, combat is mechanically rich while remaining fully deterministic.

    Architecture: Protobuf-defined skill/stats schema --> server-side SkillRegistry (per-entity) --> upgraded damage formula with seeded RNG --> per-skill cooldown tracking in entity --> projectile entities in tick loop --> UseSkill node selects skills by name/tags --> enemy templates with distinct AI trees. Client renders new combat events (CRIT/DODGE/BLOCK labels) and projectile entities.

    Tech Stack: Go 1.25, Protobuf + protojson, Godot 4.6 GDScript, gorilla/websocket

    Invariants: - Simulation MUST remain deterministic (seeded math/rand.Rand, no global random) - All 31 existing tests MUST pass at every step (nil skills/stats = backward compat) - Default behavior (no skills in build) = basic_attack only, identical to current behavior - Proto enums use ACTION_ prefix (e.g. EntityAction_ACTION_MOVE)

    package game

import pb "roguelite-server/pkg/proto"

// Skill is the runtime representation of a skill definition.
type Skill struct {
    ID             string
    Name           string
    DamageMult     float64
    Range          float64
    CooldownTicks  int
    Shape          pb.SkillShape
    DamageType     pb.DamageType
    Tags           map[string]bool
    Unstoppable    bool
    Unblockable    bool
    AOERadius      float64
    ProjectileSpeed float64
}

// SkillRegistry holds all available skills for an entity.
type SkillRegistry struct {
    byID   map[string]*Skill
    skills []*Skill
}

// DefaultBasicAttack returns the fallback melee attack every entity has.
func DefaultBasicAttack() *Skill {
    return &Skill{
        ID:            "basic_attack",
        Name:          "Basic Attack",
        DamageMult:    1.0,
        Range:         40,
        CooldownTicks: 15,
        Shape:         pb.SkillShape_SHAPE_MELEE,
        DamageType:    pb.DamageType_DAMAGE_PHYSICAL,
        Tags:          map[string]bool{"melee": true, "physical": true, "damage": true},
    }
}

// NewSkillRegistry creates a registry from protobuf skill definitions.
// Always includes basic_attack as a fallback.
func NewSkillRegistry(defs []*pb.SkillDef) *SkillRegistry {
    r := &SkillRegistry{byID: make(map[string]*Skill)}

    for _, d := range defs {
        s := skillFromDef(d)
        r.byID[s.ID] = s
        r.skills = append(r.skills, s)
    }

    // Ensure basic_attack is always present.
    if _, ok := r.byID["basic_attack"]; !ok {
        ba := DefaultBasicAttack()
        r.byID[ba.ID] = ba
        r.skills = append(r.skills, ba)
    }

    return r
}

// FindByID returns the skill with the given ID, or nil.
func (r *SkillRegistry) FindByID(id string) *Skill {
    return r.byID[id]
}

// FindByTags returns the first skill matching ALL given tags, or nil.
func (r *SkillRegistry) FindByTags(tags ...string) *Skill {
    for _, s := range r.skills {
        if s.HasAllTags(tags) {
            return s
        }
    }
    return nil
}

// All returns all registered skills.
func (r *SkillRegistry) All() []*Skill {
    return r.skills
}

// HasAllTags checks if the skill has every tag in the list.
func (s *Skill) HasAllTags(tags []string) bool {
    for _, t := range tags {
        if !s.Tags[t] {
            return false
        }
    }
    return true
}

func skillFromDef(d *pb.SkillDef) *Skill {
    tags := make(map[string]bool, len(d.Tags))
    for _, t := range d.Tags {
        tags[t] = true
    }
    return &Skill{
        ID:              d.Id,
        Name:            d.Name,
        DamageMult:      float64(d.DamageMult),
        Range:           float64(d.Range),
        CooldownTicks:   int(d.CooldownTicks),
        Shape:           d.Shape,
        DamageType:      d.DamageType,
        Tags:            tags,
        Unstoppable:     d.Unstoppable,
        Unblockable:     d.Unblockable,
        AOERadius:       float64(d.AoeRadius),
        ProjectileSpeed: float64(d.ProjectileSpeed),
    }
}

    // calcDamageV2 implements the full GDD damage formula:
// 1. Dodge check → full avoid
// 2. Block check (skip if unblockable) → reduce damage
// 3. Apply skill multiplier
// 4. Armor or resistance mitigation based on damage type
// 5. Crit roll
// 6. Minimum 1 damage
func calcDamageV2(attacker, target *entity, skill *Skill, rng *rand.Rand) DamageResult {
    // 1. Dodge check.
    if target.dodgeChance > 0 && rng.Float64() < target.dodgeChance {
        return DamageResult{WasDodged: true}
    }

    raw := float64(attacker.damage) * skill.DamageMult

    // 2. Block check (skip if unblockable).
    wasBlocked := false
    if !skill.Unblockable && target.blockChance > 0 && rng.Float64() < target.blockChance {
        raw *= (1.0 - target.blockReduction)
        wasBlocked = true
    }

    // 3. Armor or resistance mitigation.
    var defense float64
    if skill.DamageType == pb.DamageType_DAMAGE_PHYSICAL {
        defense = target.armor
    } else {
        defense = target.resistance
    }
    if defense > 0 {
        mitigation := defense / (defense + armorConstK)
        raw *= (1.0 - mitigation)
    }

    // 4. Crit roll.
    isCrit := false
    if attacker.critChance > 0 && rng.Float64() < attacker.critChance {
        raw *= attacker.critMult
        isCrit = true
    }

    // 5. Minimum 1 damage.
    final := max(int32(math.Round(raw)), 1)

    return DamageResult{
        Damage:     final,
        IsCrit:     isCrit,
        WasBlocked: wasBlocked,
    }
}

func (n *MoveTo) Evaluate(ctx *Context) (Status, *Action) {

    }

// UseSkill returns an ATTACK action using a specific skill.
// Resolution order: SkillName (exact match) → Tags (AND match) → fallback (basic_attack / old behavior).
type UseSkill struct {
    SkillName string   // exact skill ID to use
    Tags      []string // find first skill matching ALL tags
}

func (n *UseSkill) Evaluate(ctx *Context) (Status, *Action) {
    if ctx.Target == nil {
        return Failure, nil
    }

    // If no skill selection criteria → backward-compat behavior.
    if n.SkillName == "" && len(n.Tags) == 0 {
        if ctx.Self.Cooldown > 0 {
            return Failure, nil
        }
        return Success, &Action{Type: pb.EntityAction_ACTION_ATTACK, TargetID: ctx.Target.ID}
    }

    // Find skill by name or tags.
    var found *SkillInfo
    if n.SkillName != "" {
        found = findSkillByName(ctx.Skills, n.SkillName)
    } else {
        found = findSkillByTags(ctx.Skills, n.Tags)
    }

    if found == nil {
        return Failure, nil
    }

    // Check cooldown.
    if found.Cooldown > 0 {
        return Failure, nil
    }

    return Success, &Action{
        Type:     pb.EntityAction_ACTION_ATTACK,
        TargetID: ctx.Target.ID,
        SkillID:  found.ID,
    }
}

// Wait returns an IDLE action.
type Wait struct{}

func (n *Wait) Evaluate(_ *Context) (Status, *Action) {
    return Success, &Action{Type: pb.EntityAction_ACTION_IDLE}
}

func findSkillByName(skills []*SkillInfo, name string) *SkillInfo {
    for _, s := range skills {
        if s.ID == name {
            return s
        }
    }
    return nil
}

func findSkillByTags(skills []*SkillInfo, tags []string) *SkillInfo {
    for _, s := range skills {
        if hasAllTags(s.Tags, tags) {
            return s
        }
    }
    return nil
}

func hasAllTags(have []string, want []string) bool {
    set := make(map[string]bool, len(have))
    for _, t := range have {
        set[t] = true
    }
    for _, t := range want {
        if !set[t] {
            return false
        }
    }
    return true
}

            res := calcDamageV2(p.owner, e, p.skill, rng)
            p.alive = false

            if res.WasDodged {
                result.Events = append(result.Events, &pb.CombatEvent{
                    Tick: int32(tick), Type: pb.CombatEventType_DODGE,
                    SourceId: p.ownerID, TargetId: e.id,
                    SkillId: p.skill.ID, WasDodged: true,
                })
                break
            }

            e.hp -= res.Damage
            result.Events = append(result.Events, &pb.CombatEvent{
                Tick: int32(tick), Type: pb.CombatEventType_DAMAGE_DEALT,
                SourceId: p.ownerID, TargetId: e.id,
                Damage: res.Damage, RemainingHp: max(e.hp, 0),
                IsCrit: res.IsCrit, SkillId: p.skill.ID,
                WasBlocked: res.WasBlocked,
            })

            if e.hp <= 0 {
                e.alive = false
                e.action = pb.EntityAction_ACTION_DEAD
                result.Events = append(result.Events, &pb.CombatEvent{
                    Tick: int32(tick), Type: pb.CombatEventType_ENTITY_DIED,
                    SourceId: p.ownerID, TargetId: e.id,
                })
            }
            break
        }
    }
}

// ScaleStats applies floor scaling to enemy stats per GDD formula:
// HP = base * (1 + floor * 0.1), DMG = base * (1 + floor * 0.08)
func (t *EnemyTemplate) ScaleStats(floor int) (hp int32, damage int32) {
    hpMult := 1.0 + float64(floor-1)*0.1
    dmgMult := 1.0 + float64(floor-1)*0.08
    return int32(math.Round(float64(t.BaseHP) * hpMult)),
        int32(math.Round(float64(t.BaseDamage) * dmgMult))
}

// MakeEnemyConfig creates an EntityConfig from a template at the given floor.
func MakeEnemyConfig(tmpl EnemyTemplate, id string, x, y float64, floor int) EntityConfig {
    hp, dmg := tmpl.ScaleStats(floor)
    return EntityConfig{
        ID:     id,
        Type:   tmpl.Type,
        X:      x,
        Y:      y,
        HP:     hp,
        Armor:  tmpl.BaseArmor,
        Damage: dmg,
        Speed:  tmpl.BaseSpeed,
        Skills: tmpl.Skills,
        Build:  &pb.PlayerBuild{GlobalAi: tmpl.BuildAI(), TotalMemory: 100, GlobalMemory: 100},
    }
}

// SpawnEnemies returns enemy configs for a given floor.
// Floor 1-5: 3 melee. Floor 6+: 2 melee + 1 ranged.
func SpawnEnemies(floor int) []EntityConfig {
    enemies := make([]EntityConfig, 0, 3)

    if floor < 6 {
        for i := range 3 {
            enemies = append(enemies, MakeEnemyConfig(
                MeleeEnemyTemplate,
                fmt.Sprintf("enemy_%d", i),
                600, 150+float64(i)*150,
                floor,
            ))
        }
    } else {
        for i := range 2 {
            enemies = append(enemies, MakeEnemyConfig(
                MeleeEnemyTemplate,
                fmt.Sprintf("enemy_%d", i),
                600, 150+float64(i)*200,
                floor,
            ))
        }
        enemies = append(enemies, MakeEnemyConfig(
            RangedEnemyTemplate,
            "enemy_2",
            650, 300,
            floor,
        ))
    }

    return enemies
}

    Notes for Implementer¶

    • Backward compatibility is paramount. Every task must keep existing tests passing. nil skills = basic_attack only. nil stats = use EntityConfig fields directly. Zero-valued new proto fields = no behavior change.
    • Deterministic RNG. rand.New(rand.NewSource(42)) is created once per Simulate() call and passed to all functions that need randomness. NEVER use math/rand global functions.
    • Proto field numbering. New fields are added at the end of existing messages. Never reuse or reorder field numbers.
    • Proto enum string serialization. GDScript client sends proto enums as strings ("DAMAGE_PHYSICAL", "SHAPE_MELEE"). Server uses protojson which handles this automatically.
    • Projectile ownership. Projectiles store a pointer to their owner entity for damage calculation. The owner's stats at time of firing are used (not current stats). This is intentional.
    • Enemy AI uses PlayerBuild. Enemy templates generate a *pb.PlayerBuild with GlobalAi set. This reuses the same BuildExecutor() path as players. No special-casing.
    • Default AI tree doesn't select skills by name. The default UseSkill{} node (no SkillName, no Tags) falls back to basic_attack behavior. Only blueprints with explicit skill/tags parameters trigger skill selection. This means enemies with shoot_arrow need their AI tree to specify {"skill": "shoot_arrow"} in the UseSkill node parameters.
    • calcDamageV2 order of operations: dodge → block → skill mult → armor/resistance → crit → floor at 1. This matches the GDD specification.
    • Imports to add: "math/rand" in simulator.go, "strings" in builder.go, "fmt" and "math" in enemies.go.
