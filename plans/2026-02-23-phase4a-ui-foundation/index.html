    Phase 4a: UI Foundation — Implementation Plan¶

    For Claude: REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

    Goal: Full in-game UI system — main menu, settings, character profile, blueprint editor with GraphEdit, all integrated into the game loop.

    Architecture: All screens are fullscreen Control nodes managed by UIManager autoload (show/hide + navigation stack). Blueprint editor uses Godot GraphEdit for visual AI tree building, serializes to PlayerBuild Dictionary matching protojson format. Dark theme applied globally via Project Settings.

    Tech Stack: Godot 4.6 (GDScript), GraphEdit/GraphNode, ConfigFile for settings, Theme resource for styling.

    Current state: MVP with single main.tscn scene that auto-connects to server, sends hardcoded build, plays back simulation. No UI screens, no menus, no editor.

    Task 1: UIManager Autoload¶

    Files: - Create: roguelite/scripts/autoload/ui_manager.gd - Modify: roguelite/project.godot — add autoload

    Step 1: Create UIManager

    # roguelite/scripts/autoload/ui_manager.gd
extends Node
## Screen navigation manager (autoload singleton).
## Manages fullscreen UI screens with history stack for Back navigation.

signal screen_changed(screen_name: String)

var _screens: Dictionary = {}  # name → Control node
var _history: Array[String] = []
var _current: String = ""


func register_screen(screen_name: String, node: Control) -> void:
    _screens[screen_name] = node
    node.visible = false


func navigate_to(screen_name: String) -> void:
    if screen_name == _current:
        return
    if not _screens.has(screen_name):
        push_error("UIManager: unknown screen '%s'" % screen_name)
        return

    if _current != "":
        _history.push_back(_current)

    _current = screen_name
    _apply()
    screen_changed.emit(screen_name)


func go_back() -> bool:
    if _history.is_empty():
        return false
    _current = _history.pop_back()
    _apply()
    screen_changed.emit(_current)
    return true


func clear_history() -> void:
    _history.clear()


func get_current() -> String:
    return _current


func _apply() -> void:
    for name: String in _screens:
        _screens[name].visible = (name == _current)


func _unhandled_input(event: InputEvent) -> void:
    if event.is_action_pressed("ui_cancel"):
        if go_back():
            get_viewport().set_input_as_handled()

    Step 1: Create SettingsManager

    # roguelite/scripts/autoload/settings_manager.gd
extends Node
## Settings persistence via ConfigFile (autoload singleton).
## Saved to user://settings.cfg.

const SETTINGS_PATH := "user://settings.cfg"

# Video
var window_mode: int = DisplayServer.WINDOW_MODE_WINDOWED:
    set(v):
        window_mode = v
        DisplayServer.window_set_mode(v)
        _save()

var vsync: bool = true:
    set(v):
        vsync = v
        DisplayServer.window_set_vsync_mode(
            DisplayServer.VSYNC_ENABLED if v else DisplayServer.VSYNC_DISABLED
        )
        _save()

var screen_shake: bool = true:
    set(v):
        screen_shake = v
        _save()

# Audio
var master_volume: float = 1.0:
    set(v):
        master_volume = clampf(v, 0.0, 1.0)
        AudioServer.set_bus_volume_db(0, linear_to_db(master_volume))
        _save()

var music_volume: float = 0.8:
    set(v):
        music_volume = clampf(v, 0.0, 1.0)
        var idx := AudioServer.get_bus_index("Music")
        if idx >= 0:
            AudioServer.set_bus_volume_db(idx, linear_to_db(music_volume))
        _save()

var sfx_volume: float = 1.0:
    set(v):
        sfx_volume = clampf(v, 0.0, 1.0)
        var idx := AudioServer.get_bus_index("SFX")
        if idx >= 0:
            AudioServer.set_bus_volume_db(idx, linear_to_db(sfx_volume))
        _save()

# Gameplay
var default_playback_speed: float = 1.0:
    set(v):
        default_playback_speed = clampf(v, 0.5, 4.0)
        _save()

var show_ai_decisions: bool = false:
    set(v):
        show_ai_decisions = v
        _save()


func _ready() -> void:
    _load()


func _load() -> void:
    var cfg := ConfigFile.new()
    if cfg.load(SETTINGS_PATH) != OK:
        return  # First launch, use defaults

    window_mode = cfg.get_value("video", "window_mode", DisplayServer.WINDOW_MODE_WINDOWED)
    vsync = cfg.get_value("video", "vsync", true)
    screen_shake = cfg.get_value("video", "screen_shake", true)
    master_volume = cfg.get_value("audio", "master_volume", 1.0)
    music_volume = cfg.get_value("audio", "music_volume", 0.8)
    sfx_volume = cfg.get_value("audio", "sfx_volume", 1.0)
    default_playback_speed = cfg.get_value("gameplay", "default_playback_speed", 1.0)
    show_ai_decisions = cfg.get_value("gameplay", "show_ai_decisions", false)


func _save() -> void:
    var cfg := ConfigFile.new()
    cfg.set_value("video", "window_mode", window_mode)
    cfg.set_value("video", "vsync", vsync)
    cfg.set_value("video", "screen_shake", screen_shake)
    cfg.set_value("audio", "master_volume", master_volume)
    cfg.set_value("audio", "music_volume", music_volume)
    cfg.set_value("audio", "sfx_volume", sfx_volume)
    cfg.set_value("gameplay", "default_playback_speed", default_playback_speed)
    cfg.set_value("gameplay", "show_ai_decisions", show_ai_decisions)
    cfg.save(SETTINGS_PATH)

    # roguelite/scripts/ui/settings_screen.gd
extends Control
## Settings screen with Video, Audio, Gameplay tabs.

@onready var tabs: TabContainer = $VBox/TabContainer
@onready var back_btn: Button = $VBox/BackButton

# Video
@onready var window_mode_btn: OptionButton = $VBox/TabContainer/Video/WindowMode/OptionButton
@onready var vsync_check: CheckBox = $VBox/TabContainer/Video/VSync/CheckBox

# Audio
@onready var master_slider: HSlider = $VBox/TabContainer/Audio/Master/HSlider
@onready var music_slider: HSlider = $VBox/TabContainer/Audio/Music/HSlider
@onready var sfx_slider: HSlider = $VBox/TabContainer/Audio/SFX/HSlider

# Gameplay
@onready var speed_btn: OptionButton = $VBox/TabContainer/Gameplay/PlaybackSpeed/OptionButton


func _ready() -> void:
    UIManager.register_screen("settings", self)
    back_btn.pressed.connect(func(): UIManager.go_back())

    # Populate window mode options.
    window_mode_btn.add_item("Windowed", 0)
    window_mode_btn.add_item("Fullscreen", 1)
    window_mode_btn.add_item("Borderless Fullscreen", 2)

    # Populate playback speed options.
    speed_btn.add_item("0.5x", 0)
    speed_btn.add_item("1x", 1)
    speed_btn.add_item("2x", 2)
    speed_btn.add_item("4x", 3)

    _load_current()

    # Connect signals.
    window_mode_btn.item_selected.connect(_on_window_mode)
    vsync_check.toggled.connect(func(v: bool): SettingsManager.vsync = v)
    master_slider.value_changed.connect(func(v: float): SettingsManager.master_volume = v)
    music_slider.value_changed.connect(func(v: float): SettingsManager.music_volume = v)
    sfx_slider.value_changed.connect(func(v: float): SettingsManager.sfx_volume = v)
    speed_btn.item_selected.connect(_on_speed)


func _load_current() -> void:
    # Map DisplayServer window mode to OptionButton index.
    match SettingsManager.window_mode:
        DisplayServer.WINDOW_MODE_WINDOWED:
            window_mode_btn.selected = 0
        DisplayServer.WINDOW_MODE_FULLSCREEN:
            window_mode_btn.selected = 1
        DisplayServer.WINDOW_MODE_EXCLUSIVE_FULLSCREEN:
            window_mode_btn.selected = 2
        _:
            window_mode_btn.selected = 0

    vsync_check.button_pressed = SettingsManager.vsync

    master_slider.value = SettingsManager.master_volume
    music_slider.value = SettingsManager.music_volume
    sfx_slider.value = SettingsManager.sfx_volume

    # Map speed to index.
    var speed_map := {0.5: 0, 1.0: 1, 2.0: 2, 4.0: 3}
    speed_btn.selected = speed_map.get(SettingsManager.default_playback_speed, 1)


func _on_window_mode(idx: int) -> void:
    match idx:
        0: SettingsManager.window_mode = DisplayServer.WINDOW_MODE_WINDOWED
        1: SettingsManager.window_mode = DisplayServer.WINDOW_MODE_FULLSCREEN
        2: SettingsManager.window_mode = DisplayServer.WINDOW_MODE_EXCLUSIVE_FULLSCREEN


func _on_speed(idx: int) -> void:
    var speeds := [0.5, 1.0, 2.0, 4.0]
    if idx >= 0 and idx < speeds.size():
        SettingsManager.default_playback_speed = speeds[idx]

func _on_playback_tick(_tick_index: int, snapshot: Dictionary) -> void:
    var entity_states: Array = snapshot.get("entities", [])
    var seen_ids: Dictionary = {}

    for state: Dictionary in entity_states:
        var eid: String = state.get("id", "")
        seen_ids[eid] = true
        if eid in _entities:
            _entities[eid].apply_state(state)
        else:
            var etype: int = int(state.get("type", 0))
            var character: Node2D = CharacterScene.instantiate()
            entities_node.add_child(character)
            character.setup(eid, etype)
            character.apply_state(state)
            _entities[eid] = character

    var to_remove: Array = []
    for eid: String in _entities:
        if eid not in seen_ids:
            _entities[eid].queue_free()
            to_remove.append(eid)
    for eid: String in to_remove:
        _entities.erase(eid)

func _on_connection_request(from_node: StringName, from_port: int,
                            to_node: StringName, to_port: int) -> void:
    # Prevent self-loops.
    if from_node == to_node:
        return
    # Prevent duplicate input connections.
    for con: Dictionary in graph.get_connection_list():
        if con.to_node == to_node and con.to_port == to_port:
            return
    graph.connect_node(from_node, from_port, to_node, to_port)


func _on_disconnection_request(from_node: StringName, from_port: int,
                                to_node: StringName, to_port: int) -> void:
    graph.disconnect_node(from_node, from_port, to_node, to_port)


func _on_delete_nodes_request(nodes: Array[StringName]) -> void:
    for node_name: StringName in nodes:
        var node := graph.get_node_or_null(NodePath(node_name))
        if node is BlueprintGraphNode:
            # Remove all connections to/from this node.
            for con: Dictionary in graph.get_connection_list():
                if con.from_node == node_name or con.to_node == node_name:
                    graph.disconnect_node(con.from_node, con.from_port,
                                          con.to_node, con.to_port)
            node.queue_free()
    await get_tree().process_frame
    _update_status()

func serialize_to_build() -> Dictionary:
    ## Convert the GraphEdit graph into a PlayerBuild Dictionary (protojson format).
    var connections := graph.get_connection_list()

    # Find root nodes (nodes with no incoming connections).
    var has_input: Dictionary = {}
    for con: Dictionary in connections:
        has_input[String(con.to_node)] = true

    var roots: Array[BlueprintGraphNode] = []
    for child in graph.get_children():
        if child is BlueprintGraphNode:
            if not has_input.has(child.name):
                roots.append(child)

    if roots.is_empty():
        push_warning("Blueprint has no root nodes!")
        return {}

    # Build tree from the first root (global AI).
    var global_ai := _serialize_node(roots[0], connections)

    # Get current profile stats/skills or use defaults.
    var profile_screen = UIManager._screens.get("profile")
    var current_build: Dictionary = {}
    if profile_screen and profile_screen.has_method("get_build"):
        current_build = profile_screen.get_build()

    return {
        "globalAi": global_ai,
        "totalMemory": current_build.get("totalMemory", 20),
        "globalMemory": current_build.get("globalMemory", 20),
        "skills": current_build.get("skills", [
            {
                "id": "basic_attack", "name": "Basic Attack",
                "damageMult": 1.0, "range": 40.0, "cooldownTicks": 15,
                "shape": "SHAPE_MELEE", "damageType": "DAMAGE_PHYSICAL",
                "tags": ["melee", "physical", "damage"]
            }
        ]),
        "stats": current_build.get("stats", {
            "maxHp": 100, "baseDamage": 15, "armor": 10.0,
            "resistance": 0.0, "speed": 3.0, "critChance": 0.05,
            "critMult": 1.5, "dodgeChance": 0.0,
            "blockChance": 0.0, "blockReduction": 0.0
        })
    }


func _serialize_node(gnode: BlueprintGraphNode, connections: Array) -> Dictionary:
    var result := {"type": gnode.node_type}

    # Add parameters if any.
    if not gnode.node_params.is_empty():
        var params := {}
        for key: String in gnode.node_params:
            var val = gnode.node_params[key]
            if val is float:
                params[key] = str(val)
            elif val is String and val != "":
                params[key] = val
        if not params.is_empty():
            result["parameters"] = params

    # Find children connected to this node's output ports.
    var children_by_port: Dictionary = {}
    for con: Dictionary in connections:
        if String(con.from_node) == gnode.name:
            var target_name := String(con.to_node)
            var target_node := graph.get_node_or_null(NodePath(target_name))
            if target_node is BlueprintGraphNode:
                children_by_port[con.from_port] = target_node

    if not children_by_port.is_empty():
        var sorted_ports := children_by_port.keys()
        sorted_ports.sort()
        var children_array := []
        for port: int in sorted_ports:
            children_array.append(_serialize_node(children_by_port[port], connections))
        result["children"] = children_array

    return result

    Task 11: Dark Theme¶

    Files: - Create: roguelite/assets/themes/dark_theme.tres (via Godot editor or code) - Modify: roguelite/project.godot — set theme

    Step 1: Create theme programmatically

    Rather than manually editing .tres, create a script that generates and saves the theme at editor time. But for simplicity, set theme overrides in main.gd:

    Add to main.gd _ready():

    func _ready() -> void:
    _apply_dark_theme()
    await get_tree().process_frame
    UIManager.navigate_to("main_menu")

func _apply_dark_theme() -> void:
    # Background color for all panels.
    var bg := StyleBoxFlat.new()
    bg.bg_color = Color(0.1, 0.1, 0.12)

    # Button styles.
    var btn_normal := StyleBoxFlat.new()
    btn_normal.bg_color = Color(0.2, 0.2, 0.25)
    btn_normal.corner_radius_top_left = 4
    btn_normal.corner_radius_top_right = 4
    btn_normal.corner_radius_bottom_left = 4
    btn_normal.corner_radius_bottom_right = 4
    btn_normal.border_width_bottom = 1
    btn_normal.border_width_left = 1
    btn_normal.border_width_right = 1
    btn_normal.border_width_top = 1
    btn_normal.border_color = Color(0.35, 0.35, 0.4)

    var btn_hover := btn_normal.duplicate()
    btn_hover.bg_color = Color(0.3, 0.3, 0.38)

    var btn_pressed := btn_normal.duplicate()
    btn_pressed.bg_color = Color(0.15, 0.15, 0.2)

    var btn_disabled := btn_normal.duplicate()
    btn_disabled.bg_color = Color(0.15, 0.15, 0.18)

    var t := Theme.new()
    t.set_stylebox("panel", "PanelContainer", bg)
    t.set_stylebox("normal", "Button", btn_normal)
    t.set_stylebox("hover", "Button", btn_hover)
    t.set_stylebox("pressed", "Button", btn_pressed)
    t.set_stylebox("disabled", "Button", btn_disabled)
    t.set_color("font_color", "Button", Color(0.9, 0.9, 0.9))
    t.set_color("font_hover_color", "Button", Color(1.0, 1.0, 1.0))
    t.set_color("font_disabled_color", "Button", Color(0.5, 0.5, 0.5))
    t.set_color("font_color", "Label", Color(0.85, 0.85, 0.85))
    t.set_color("font_color", "LineEdit", Color(0.9, 0.9, 0.9))

    # Tab styles.
    var tab_selected := StyleBoxFlat.new()
    tab_selected.bg_color = Color(0.25, 0.25, 0.3)
    var tab_unselected := StyleBoxFlat.new()
    tab_unselected.bg_color = Color(0.15, 0.15, 0.18)
    t.set_stylebox("tab_selected", "TabContainer", tab_selected)
    t.set_stylebox("tab_unselected", "TabContainer", tab_unselected)

    # Apply to root — cascades to all children.
    get_tree().root.theme = t

    Alternatively, if you prefer a .tres file, create it through the Godot editor's Theme Editor and set it in Project Settings → GUI → Theme → Custom.
